cmake_minimum_required(VERSION 3.16)
project(pcsc-rw5100 VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
include(CTest)
include(GNUInstallDirs)
find_package(Threads REQUIRED)
find_package(PkgConfig QUIET)
set(RW5100_SOURCE_DIR "" CACHE PATH "Optional librw5100 source tree for development builds")
if(RW5100_SOURCE_DIR)
  set(saved_testing ${BUILD_TESTING})
  set(BUILD_TESTING OFF)
  add_subdirectory(${RW5100_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR}/librw5100 EXCLUDE_FROM_ALL)
  set(BUILD_TESTING ${saved_testing})
  set_target_properties(rw5100 PROPERTIES POSITION_INDEPENDENT_CODE ON)
  set(RW_TARGET rw5100::rw5100)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(RW5100 REQUIRED IMPORTED_TARGET librw5100)
  set(RW_TARGET PkgConfig::RW5100)
endif()
if(APPLE)
  find_library(PCSC_FRAMEWORK PCSC REQUIRED)
  set(PCSC_TARGET ${PCSC_FRAMEWORK})
  set(IFD_PLATFORM MacOS)
  set(IFD_INSTALL_DEFAULT /usr/local/libexec/SmartCardServices/drivers)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(PCSC REQUIRED IMPORTED_TARGET libpcsclite)
  set(PCSC_TARGET PkgConfig::PCSC)
  set(IFD_PLATFORM Linux)
  pkg_get_variable(IFD_INSTALL_DEFAULT libpcsclite usbdropdir)
  if(NOT IFD_INSTALL_DEFAULT)
    set(IFD_INSTALL_DEFAULT ${CMAKE_INSTALL_FULL_LIBDIR}/pcsc/drivers)
  endif()
endif()
set(PCSC_DRIVER_DIR "${IFD_INSTALL_DEFAULT}" CACHE PATH "System PC/SC driver bundle directory")
set(BUNDLE_DIR ${CMAKE_CURRENT_BINARY_DIR}/ifd-rw5100.bundle)
file(MAKE_DIRECTORY ${BUNDLE_DIR}/Contents)
configure_file(src/Info.plist.in ${BUNDLE_DIR}/Contents/Info.plist @ONLY)
add_library(ifd-rw5100 MODULE src/ifd.c)
target_include_directories(ifd-rw5100 PRIVATE src)
target_link_libraries(ifd-rw5100 PRIVATE ${RW_TARGET} ${PCSC_TARGET} Threads::Threads)
set_target_properties(ifd-rw5100 PROPERTIES PREFIX "" SUFFIX "" C_VISIBILITY_PRESET hidden
  LIBRARY_OUTPUT_DIRECTORY ${BUNDLE_DIR}/Contents/${IFD_PLATFORM})
# Export only the resource-manager ABI; no reader-library symbols leak out.
if(APPLE)
  target_link_options(ifd-rw5100 PRIVATE "LINKER:-exported_symbols_list,${CMAKE_CURRENT_SOURCE_DIR}/src/exports.macos")
else()
  target_link_options(ifd-rw5100 PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/src/exports.map")
endif()
target_compile_options(ifd-rw5100 PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(APPLE)
  find_program(CODESIGN codesign REQUIRED)
  add_custom_command(TARGET ifd-rw5100 POST_BUILD
    COMMAND ${CODESIGN} --force --sign - ${BUNDLE_DIR}
    COMMENT "Ad-hoc sign the complete IFD bundle for local testing")
endif()
install(DIRECTORY ${BUNDLE_DIR} DESTINATION ${PCSC_DRIVER_DIR} COMPONENT pcsc)
add_executable(rw5100-pcsc-check tools/pcsc-check.c)
target_link_libraries(rw5100-pcsc-check PRIVATE ${PCSC_TARGET})
if(BUILD_TESTING)
  add_executable(test_ifd tests/test_ifd.c src/ifd.c)
  target_include_directories(test_ifd PRIVATE src)
  target_link_libraries(test_ifd PRIVATE ${RW_TARGET} ${PCSC_TARGET} Threads::Threads)
  add_test(NAME ifd_contract COMMAND test_ifd)
  add_executable(ifd-check tools/ifd-check.c)
  target_include_directories(ifd-check PRIVATE src)
  target_link_libraries(ifd-check PRIVATE ${PCSC_TARGET} ${CMAKE_DL_LIBS})
endif()
